async function fetchData() { const rawInput = document.getElementById('searchInput').value.trim().toLowerCase(); const err = document.getElementById('errMsg'); const btn = document.getElementById('btnSearch'); err.style.display = 'none'; if (!rawInput) { err.innerText = "يرجى كتابة البريد الإلكتروني الخاص بك أولاً."; err.style.display = 'block'; return; } const idMatch = rawInput.match(/\d+/); const studentNumber = idMatch ? idMatch[0] : ""; btn.innerText = "جاري جلب النتائج..."; btn.disabled = true; try { const snap = await db.collection("alkfaeat_results").get(); studentData = []; snap.forEach(doc => { const d = doc.data(); const docEmail = String(d.email || "").toLowerCase().trim(); const docName = String(d.name || "").toLowerCase().trim(); // مطابقة شاملة ومرنة تضمن جلب كافة اختبارات الطالب const isEmailMatch = docEmail && (docEmail === rawInput || docEmail.includes(rawInput) || rawInput.includes(docEmail)); const isNumberMatch = studentNumber && (docEmail.includes(studentNumber) || docName.includes(studentNumber)); // إذا لم يتم تسجيل البريد في بعض السجلات القديمة، نتيح البحث باسم الطالب أو تطابق تام if (isEmailMatch || isNumberMatch) { studentData.push(d); } }); // ترتيب الاختبارات زمنياً من الأقدم للأحدث studentData.sort((a, b) => { let tA = a.time ? (a.time.toDate ? a.time.toDate() : new Date(a.time)) : 0; let tB = b.time ? (b.time.toDate ? b.time.toDate() : new Date(b.time)) : 0; return tA - tB; }); if (studentData.length === 0) { err.innerText = "لم نتمكن من العثور على أي نتائج مسجلة بهذا البريد الإلكتروني. يرجى التأكد من كتابة الإيميل كما هو مُدخل أثناء حل الاختبار."; err.style.display = 'block'; btn.innerText = "عرض النتائج"; btn.disabled = false; return; } document.getElementById('searchSection').style.display = 'none'; document.getElementById('dashboardSection').style.display = 'block'; renderUI(rawInput); } catch (e) { console.error(e); err.innerText = "خطأ في الاتصال بالسحابة: " + e.message; err.style.display = 'block'; } finally { btn.innerText = "عرض النتائج"; btn.disabled = false; } }
